def reverse_num(n):
    rev = 0
    while n > 0:
        rev = rev * 10 + (n % 10)
        n //= 10
    return rev


def digit_count(n):
    count = 0
    while n > 0:
        count += 1
        n //= 10
    return count


def is_mirror_pair(a, b):
    if a <= 0 or b <= 0:
        return False
    return reverse_num(a) == b and digit_count(a) == digit_count(b)


def is_mirror_list(numbers):
    left = 0
    right = len(numbers) - 1

    while left <= right:
        if not is_mirror_pair(numbers[left], numbers[right]):
            return False
        left += 1
        right -= 1

    return True
